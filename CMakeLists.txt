cmake_minimum_required(VERSION 3.18)
project(bevfusion_c VERSION 0.2.0 LANGUAGES C)

set(BF_ENABLE_CUDA "AUTO" CACHE STRING "Build CUDA backend: AUTO, ON, or OFF")
set_property(CACHE BF_ENABLE_CUDA PROPERTY STRINGS AUTO ON OFF)
option(BF_ENABLE_CUDA_VENDOR "Build the optional cuDNN/cuBLAS comparison executable" OFF)
set(BF_CUDA_ARCHS "" CACHE STRING "CUDA architectures, for example 87 or 89")
set(BF_ENABLE_BLAS "AUTO" CACHE STRING "Use CBLAS/Accelerate CPU kernels: AUTO, ON, or OFF")
set_property(CACHE BF_ENABLE_BLAS PROPERTY STRINGS AUTO ON OFF)
set(BF_ENABLE_OPENMP "AUTO" CACHE STRING "Use OpenMP CPU kernels: AUTO, ON, or OFF")
set_property(CACHE BF_ENABLE_OPENMP PROPERTY STRINGS AUTO ON OFF)
option(BF_NATIVE "Enable -march=native for a machine-local CPU build" OFF)
option(BF_BUILD_TESTS "Build dependency-free unit tests" ON)
option(BF_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

foreach(mode BF_ENABLE_CUDA BF_ENABLE_BLAS BF_ENABLE_OPENMP)
  if(NOT ${mode} MATCHES "^(AUTO|ON|OFF)$")
    message(FATAL_ERROR "${mode} must be AUTO, ON, or OFF")
  endif()
endforeach()

set(BF_RUNTIME_SOURCES
  src/runtime.c src/model.c src/kernels_ref.c src/voxel.c src/swin.c
  src/swin_backbone.c src/image_fpn.c src/depth_raster.c src/depth_head.c
  src/lss.c src/lss_downsample.c src/lidar_backbone.c src/bev_stage.c
  src/transfusion.c src/transfusion_decoder.c)
set(BF_CLI_SOURCES src/frame.c src/tui.c)

add_library(bevfusion_runtime STATIC ${BF_RUNTIME_SOURCES})
target_include_directories(bevfusion_runtime PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_features(bevfusion_runtime PUBLIC c_std_11)

set(BF_BLAS_AVAILABLE FALSE)
if(NOT BF_ENABLE_BLAS STREQUAL "OFF")
  if(APPLE)
    find_library(BF_ACCELERATE_FRAMEWORK Accelerate)
    if(BF_ACCELERATE_FRAMEWORK)
      set(BF_BLAS_AVAILABLE TRUE)
      target_compile_definitions(bevfusion_runtime PRIVATE BF_WITH_ACCELERATE)
      target_link_libraries(bevfusion_runtime PUBLIC ${BF_ACCELERATE_FRAMEWORK})
    endif()
  else()
    include(CheckIncludeFile)
    check_include_file(cblas.h BF_HAVE_CBLAS_H)
    if(BF_ENABLE_BLAS STREQUAL "AUTO")
      set(BLA_VENDOR OpenBLAS)
    endif()
    find_package(BLAS QUIET)
    if(BF_HAVE_CBLAS_H AND BLAS_FOUND)
      set(BF_BLAS_AVAILABLE TRUE)
      target_compile_definitions(bevfusion_runtime PRIVATE BF_WITH_CBLAS)
      target_link_libraries(bevfusion_runtime PUBLIC BLAS::BLAS)
    endif()
  endif()
  if(BF_ENABLE_BLAS STREQUAL "ON" AND NOT BF_BLAS_AVAILABLE)
    message(FATAL_ERROR "BLAS was requested but CBLAS or Accelerate was not found")
  endif()
endif()

set(BF_OPENMP_AVAILABLE FALSE)
if(NOT BF_ENABLE_OPENMP STREQUAL "OFF")
  find_package(OpenMP QUIET COMPONENTS C)
  if(OpenMP_C_FOUND)
    set(BF_OPENMP_AVAILABLE TRUE)
    target_compile_definitions(bevfusion_runtime PRIVATE BF_WITH_OPENMP)
    target_link_libraries(bevfusion_runtime PUBLIC OpenMP::OpenMP_C)
  elseif(BF_ENABLE_OPENMP STREQUAL "ON")
    message(FATAL_ERROR "OpenMP was requested but the C compiler/runtime was not found")
  endif()
endif()

add_executable(bevfusion src/main.c ${BF_CLI_SOURCES})
target_link_libraries(bevfusion PRIVATE bevfusion_runtime m)

foreach(target bevfusion_runtime bevfusion)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(BF_NATIVE)
      target_compile_options(${target} PRIVATE -march=native)
    endif()
  endif()
endforeach()

set(BF_CUDA_AVAILABLE FALSE)
if(NOT BF_ENABLE_CUDA STREQUAL "OFF")
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    find_package(CUDAToolkit QUIET)
    if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 12.0 AND CUDAToolkit_FOUND)
      set(BF_CUDA_AVAILABLE TRUE)
    endif()
  endif()
  if(BF_ENABLE_CUDA STREQUAL "ON" AND NOT BF_CUDA_AVAILABLE)
    message(FATAL_ERROR "CUDA was requested but CUDA Toolkit 12.0 or newer was not found. Set CUDAToolkit_ROOT, or configure with -DBF_ENABLE_CUDA=OFF.")
  elseif(NOT BF_CUDA_AVAILABLE)
    message(STATUS "CUDA backend disabled: CUDA Toolkit 12.0 or newer not found")
  endif()
endif()
if(BF_ENABLE_CUDA_VENDOR AND NOT BF_CUDA_AVAILABLE)
  message(FATAL_ERROR "BF_ENABLE_CUDA_VENDOR=ON also requires BF_ENABLE_CUDA and CUDA Toolkit 12.0 or newer")
endif()

if(BF_CUDA_AVAILABLE)
  set(BF_CUDA_SOURCES src/cuda_runtime.cu src/cuda_depth_raster.cu
    src/cuda_voxel.cu src/cuda_lidar.cu src/cuda_swin.cu src/cuda_camera.cu
    src/cuda_lss.cu src/cuda_bev_stage.cu src/cuda_transfusion.cu
    src/cuda_ops.cu)
  add_executable(bevfusion_cuda src/main.c ${BF_CLI_SOURCES} ${BF_CUDA_SOURCES})
  target_compile_definitions(bevfusion_cuda PRIVATE BF_WITH_CUDA)
  target_include_directories(bevfusion_cuda PRIVATE include)
  target_compile_features(bevfusion_cuda PRIVATE c_std_11 cuda_std_14)
  target_link_libraries(bevfusion_cuda PRIVATE bevfusion_runtime CUDA::cudart m)
  set_target_properties(bevfusion_cuda PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
  if(BF_CUDA_ARCHS)
    set_target_properties(bevfusion_cuda PROPERTIES CUDA_ARCHITECTURES "${BF_CUDA_ARCHS}")
  else()
    set_target_properties(bevfusion_cuda PROPERTIES CUDA_ARCHITECTURES "75")
  endif()

  if(BF_ENABLE_CUDA_VENDOR)
    find_path(CUDNN_INCLUDE_DIR cudnn.h HINTS $ENV{CUDNN_ROOT}
      ${CUDAToolkit_INCLUDE_DIRS} PATH_SUFFIXES include)
    find_library(CUDNN_LIBRARY cudnn HINTS $ENV{CUDNN_ROOT}
      PATH_SUFFIXES lib lib64 lib/aarch64-linux-gnu)
    if(NOT CUDNN_INCLUDE_DIR OR NOT CUDNN_LIBRARY)
      message(FATAL_ERROR "BF_ENABLE_CUDA_VENDOR=ON requires cuDNN headers and library")
    endif()
    add_executable(bevfusion_cuda_vendor src/main.c ${BF_CLI_SOURCES} ${BF_CUDA_SOURCES})
    target_compile_definitions(bevfusion_cuda_vendor PRIVATE BF_WITH_CUDA BF_CUDA_VENDOR)
    target_include_directories(bevfusion_cuda_vendor PRIVATE include ${CUDNN_INCLUDE_DIR})
    target_compile_features(bevfusion_cuda_vendor PRIVATE c_std_11 cuda_std_14)
    target_link_libraries(bevfusion_cuda_vendor PRIVATE bevfusion_runtime
      CUDA::cudart CUDA::cublas ${CUDNN_LIBRARY} m)
    set_target_properties(bevfusion_cuda_vendor PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
    if(BF_CUDA_ARCHS)
      set_target_properties(bevfusion_cuda_vendor PROPERTIES CUDA_ARCHITECTURES "${BF_CUDA_ARCHS}")
    else()
      set_target_properties(bevfusion_cuda_vendor PROPERTIES CUDA_ARCHITECTURES "75")
    endif()
  endif()

  if(BF_BUILD_TESTS)
    add_executable(test_cuda_ops tests/test_cuda_ops.cu src/cuda_ops.cu)
    target_include_directories(test_cuda_ops PRIVATE include)
    target_compile_features(test_cuda_ops PRIVATE cuda_std_14)
    target_link_libraries(test_cuda_ops PRIVATE CUDA::cudart)
    set_target_properties(test_cuda_ops PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
    if(BF_CUDA_ARCHS)
      set_target_properties(test_cuda_ops PROPERTIES CUDA_ARCHITECTURES "${BF_CUDA_ARCHS}")
    else()
      set_target_properties(test_cuda_ops PROPERTIES CUDA_ARCHITECTURES "75")
    endif()
    add_test(NAME test_cuda_ops COMMAND test_cuda_ops)
  endif()
endif()

if(BF_BUILD_TESTS)
  include(CTest)
  add_executable(test_model tests/test_model.c src/model.c)
  add_executable(test_frame tests/test_frame.c src/frame.c)
  add_executable(test_tui tests/test_tui.c src/tui.c)
  add_executable(test_cpu_dispatch tests/test_cpu_dispatch.c)
  target_link_libraries(test_cpu_dispatch PRIVATE bevfusion_runtime)
  foreach(test_target test_model test_frame test_tui test_cpu_dispatch)
    target_include_directories(${test_target} PRIVATE include)
    target_compile_features(${test_target} PRIVATE c_std_11)
    target_link_libraries(${test_target} PRIVATE m)
    add_test(NAME ${test_target} COMMAND ${test_target})
  endforeach()
endif()

# A sanitized static library requires every final executable that consumes it
# to link the sanitizer runtime. Apply the flags through one interface target so
# they propagate to the CLI and tests instead of relying on target order.
if(BF_SANITIZERS)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
    add_library(bf_sanitizers INTERFACE)
    target_compile_options(bf_sanitizers INTERFACE
      -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bf_sanitizers INTERFACE -fsanitize=address,undefined)
    target_link_libraries(bevfusion_runtime PUBLIC bf_sanitizers)
    if(BF_BUILD_TESTS)
      target_link_libraries(test_model PRIVATE bf_sanitizers)
      target_link_libraries(test_frame PRIVATE bf_sanitizers)
      target_link_libraries(test_tui PRIVATE bf_sanitizers)
    endif()
  else()
    message(FATAL_ERROR "BF_SANITIZERS is only supported with GNU or Clang-family C compilers")
  endif()
endif()

include(GNUInstallDirs)
install(TARGETS bevfusion bevfusion_runtime
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(TARGET bevfusion_cuda)
  install(TARGETS bevfusion_cuda RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(TARGET bevfusion_cuda_vendor)
  install(TARGETS bevfusion_cuda_vendor RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()

message(STATUS "BEVFusion.c CPU backend: ON")
message(STATUS "BEVFusion.c CPU BLAS: ${BF_BLAS_AVAILABLE}")
message(STATUS "BEVFusion.c CPU OpenMP: ${BF_OPENMP_AVAILABLE}")
message(STATUS "BEVFusion.c CUDA backend: ${BF_CUDA_AVAILABLE}")
message(STATUS "BEVFusion.c CUDA vendor comparison: ${BF_ENABLE_CUDA_VENDOR}")
